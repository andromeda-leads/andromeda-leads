Подключение облачного сохранения 🌸

Нужно один раз, около пяти минут, с компьютера. После этого данные тренировок будут сохраняться в вашей Google-таблице, а приложение сможет показывать их на любом устройстве.

  1. Откройте sheets.new - создастся новая таблица. Назовите её «Хоккей: 8 месяцев».
  2. В меню таблицы: Расширения → Apps Script.
  3. В редакторе удалите весь текст и вставьте код ниже (кнопка «Скопировать код»).
  4. Развернуть → Новое развёртывание → тип «Веб-приложение». «Выполнять от имени» - ваш аккаунт, «У кого есть доступ» - Все. Нажмите «Развернуть», при предупреждении Google - «Дополнительно» → «Перейти» → «Разрешить».
  5. Скопируйте адрес веб-приложения (заканчивается на /exec) и пришлите его ассистенту - он пропишет адрес в приложении, и синхронизация заработает сама.

Код для Apps Script

/**
 * Хоккей: 8 месяцев — приём и хранение данных приложения.
 *
 * Как поставить (5 минут, один раз):
 *  1. Откройте https://sheets.new — создастся новая Google-таблица. Назовите её «Хоккей: 8 месяцев».
 *  2. В таблице: меню «Расширения» → «Apps Script».
 *  3. Удалите то, что в редакторе, вставьте весь этот текст.
 *  4. Нажмите «Развернуть» → «Новое развёртывание» → тип «Веб-приложение»:
 *       — «Выполнять от имени»: Я (ваш аккаунт)
 *       — «У кого есть доступ»: Все
 *     Нажмите «Развернуть» и разрешите доступ (появится предупреждение Google — «Дополнительно» → «Перейти»).
 *  5. Скопируйте адрес веб-приложения (заканчивается на /exec) — он нужен в приложении.
 *
 * Код доступа ниже (TOKEN) должен совпадать с кодом в приложении.
 */

var TOKEN = 'hockey-8m-2026';

function doGet(e) {
  var p = (e && e.parameter) || {};
  if (p.token !== TOKEN) return ok({ ok: false, error: 'bad_token' });

  if (p.mode === 'get') {
    var sh = ensureSheet('state', ['key', 'json', 'updated']);
    var data = sh.getDataRange().getValues();
    for (var i = 1; i < data.length; i++) {
      if (data[i][0] === 'main') {
        return ok({ ok: true, updated: String(data[i][2]), state: JSON.parse(data[i][1] || 'null') });
      }
    }
    return ok({ ok: true, state: null });
  }

  if (p.mode === 'history') {
    var h = ensureSheet('history', ['когда', 'опыт XP', 'уровень', 'тренировок выполнено', 'пропусков', 'дней болезни', 'записей в дневнике', 'комментарий']);
    var vals = h.getDataRange().getValues();
    return ok({ ok: true, rows: vals.slice(1).slice(-60) });
  }

  return ok({ ok: true, message: 'hockey backend alive', time: new Date().toISOString() });
}

function doPost(e) {
  var body;
  try { body = JSON.parse((e && e.postData && e.postData.contents) || '{}'); }
  catch (err) { return ok({ ok: false, error: 'bad_json' }); }
  if (body.token !== TOKEN) return ok({ ok: false, error: 'bad_token' });
  var st = body.state;
  if (!st) return ok({ ok: false, error: 'no_state' });

  var sh = ensureSheet('state', ['key', 'json', 'updated']);
  var data = sh.getDataRange().getValues();
  var now = new Date();
  var json = JSON.stringify(st);
  var written = false;
  for (var i = 1; i < data.length; i++) {
    if (data[i][0] === 'main') {
      sh.getRange(i + 1, 2).setValue(json);
      sh.getRange(i + 1, 3).setValue(now);
      written = true;
      break;
    }
  }
  if (!written) sh.appendRow(['main', json, now]);

  var metrics = summary(st);
  ensureSheet('history', ['когда', 'опыт XP', 'уровень', 'тренировок выполнено', 'пропусков', 'дней болезни', 'записей в дневнике', 'комментарий'])
    .appendRow([now, metrics.xp, metrics.level, metrics.done, metrics.skipped, metrics.sick, metrics.diary, '']);

  return ok({ ok: true, saved: now.toISOString(), xp: metrics.xp, done: metrics.done });
}

/* ---------- разбор состояния ---------- */
function summary(st) {
  var checks = st.checks || {}, diary = st.diary || {}, tests = st.tests || {};
  var done = 0, skipped = 0, partial = 0;
  for (var d in checks) {
    var rec = checks[d];
    if (!rec) continue;
    if (rec.status === 'skipped') { skipped++; continue; }
    var items = rec.items || {}, n = 0;
    for (var k in items) if (items[k]) n++;
    if (n > 0) done++;
  }
  var testsN = 0;
  for (var t in tests) testsN += (tests[t] || []).length;
  var xp = st.xp || 0;
  return {
    xp: xp,
    level: 1 + Math.floor(xp / 150),
    done: done,
    skipped: skipped,
    sick: (st.sick || []).length,
    diary: Object.keys(diary).length,
    tests: testsN
  };
}

/* ---------- служебное ---------- */
function ensureSheet(name, header) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sh = ss.getSheetByName(name);
  if (!sh) sh = ss.insertSheet(name);
  if (sh.getLastRow() === 0) sh.appendRow(header);
  sh.getRange(1, 1, 1, header.length).setFontWeight('bold');
  return sh;
}

function ok(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj)).setMimeType(ContentService.MimeType.JSON);
}

function testRun() {
  var st = { checks: { '2026-10-19': { items: { a: true } } }, diary: {}, tests: {}, sick: [], xp: 120 };
  Logger.log(doPost({ postData: { contents: JSON.stringify({ token: TOKEN, state: st }) } }).getContent());
  Logger.log(doGet({ parameter: { token: TOKEN, mode: 'get' } }).getContent());
  Logger.log(doGet({ parameter: { token: TOKEN, mode: 'history' } }).getContent());
}
